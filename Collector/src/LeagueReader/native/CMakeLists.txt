cmake_minimum_required(VERSION 3.16)
project(lc_reader VERSION 0.1.0 LANGUAGES CXX)

add_library(lc_reader SHARED src/lc_reader.cpp src/win32_reader.cpp)
target_compile_features(lc_reader PUBLIC cxx_std_17)
target_include_directories(lc_reader PUBLIC include)
target_compile_definitions(lc_reader PRIVATE LC_READER_BUILD)
set_target_properties(lc_reader PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(MSVC)
    target_compile_options(lc_reader PRIVATE /W4 /permissive- /EHsc)
else()
    target_compile_options(lc_reader PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS lc_reader RUNTIME DESTINATION bin LIBRARY DESTINATION lib)
install(FILES include/lc_reader.h DESTINATION include)
